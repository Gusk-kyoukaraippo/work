Option Explicit

Public Sub RunStartupNativeTests()
    On Error GoTo Failed
    If ThisWorkbook.Worksheets.Count <> 1 Then Err.Raise 5, , "Use an uninitialized study master only"
    Dim n As Long, i As Long, k As Long, bytes As Variant, passed As Long
    If GateCrc32TableBytes(Array(49, 50, 51, 52, 53, 54, 55, 56, 57)) <> "CBF43926" Then Err.Raise 5, , "known vector"
    If GateCrc32TableBytes(Array()) <> "00000000" Then Err.Raise 5, , "empty vector"
    For n = 0 To 255
        bytes = Array(n)
        If GateCrc32TableBytes(bytes) <> ReferenceCrc(bytes) Then Err.Raise 5, , "single byte"
        passed = passed + 1
    Next n
    For k = 1 To 12
        n = k * k * 97
        ReDim bytes(0 To n - 1)
        For i = 0 To n - 1
            bytes(i) = (i * 73 + k * 19 + CLng(Fix(i / 251))) Mod 256
        Next i
        If GateCrc32TableBytes(bytes) <> ReferenceCrc(bytes) Then Err.Raise 5, , "generated vector"
        passed = passed + 1
    Next k
    Dim ws As Object
    Set ws = ThisWorkbook.Worksheets.Add
    ws.Name = GATE_META_SHEET
    ws.Cells(1, 1).Value2 = "key": ws.Cells(1, 2).Value2 = "value"
    ws.Cells(2, 1).Value2 = "activeSessionId": ws.Cells(2, 2).Value2 = "before"
    ws.Cells(3, 1).Value2 = "lastSaveAt": ws.Cells(3, 2).Value2 = 46000.5
    ws.Cells(4, 1).Value2 = "activeSessionId": ws.Cells(4, 2).Value2 = "duplicate"
    Dim dateText As String
    dateText = GateMetaGet("lastSaveAt")
    ThisWorkbook.Names.Add Name:="_GateStartupProfile", RefersTo:="=" & Chr$(34) & "minimal" & Chr$(34)
    GateStartupBegin
    If Not GateStartupFast() Then Err.Raise 5, , "scope disabled"
    If GateMetaGet("activeSessionId") <> "before" Then Err.Raise 5, , "first duplicate semantics"
    If GateMetaGet("ActiveSessionId", "missing") <> "missing" Then Err.Raise 5, , "key case changed"
    If GateMetaGet("missing", "default") <> "default" Then Err.Raise 5, , "missing default"
    If GateMetaGet("lastSaveAt") <> dateText Then Err.Raise 5, , "date semantics"
    GateMetaSet "activeSessionId", "after"
    If GateMetaGet("activeSessionId") <> "after" Then Err.Raise 5, , "write invalidation"
    GateMetaSet "newField", "added"
    If GateMetaGet("newField") <> "added" Then Err.Raise 5, , "append invalidation"
    GateStartupEnd
    If GateStartupFast() Then Err.Raise 5, , "scope leaked"
    ws.Cells(2, 2).Value2 = "next"
    GateStartupBegin
    If GateMetaGet("activeSessionId") <> "next" Then Err.Raise 5, , "next operation stale"
    Dim json As String
    json = "{" & Chr$(34) & "dataVersion" & Chr$(34) & ":2}"
    If GateStartupConfigValue(json, "dataVersion", "long") <> 2 Then Err.Raise 5, , "config parse"
    GateStartupEnd
    GateStartupBegin
    json = "{" & Chr$(34) & "dataVersion" & Chr$(34) & ":3}"
    If GateStartupConfigValue(json, "dataVersion", "long") <> 3 Then Err.Raise 5, , "config leaked"
    GateStartupEnd
    CleanupStudyTests
    ThisWorkbook.Worksheets(GATE_PANEL_SHEET).Unprotect GATE_PROTECT_PASSWORD
    ThisWorkbook.Worksheets(GATE_PANEL_SHEET).Range("B2").Value2 = "Excelゲート"
    GateApplyProtection
    MsgBox "PASS: 270 CRC vectors; metadata dates/duplicates/write invalidation; operation cache reset", vbInformation, "Startup native tests"
    Exit Sub
Failed:
    Dim problem As String
    problem = Err.Description
    CleanupStudyTests
    MsgBox "FAIL: " & problem, vbCritical, "Startup native tests"
End Sub

Private Sub CleanupStudyTests()
    On Error Resume Next
    GateStartupEnd
    ThisWorkbook.Names("_GateStartupProfile").Delete
    Dim oldAlerts As Boolean
    oldAlerts = Application.DisplayAlerts
    Application.DisplayAlerts = False
    ThisWorkbook.Worksheets(GATE_META_SHEET).Delete
    Application.DisplayAlerts = oldAlerts
End Sub

Private Function ReferenceCrc(ByVal bytes As Variant) As String
    Dim crc As Long, i As Long, bit As Long, low As Boolean
    crc = &HFFFFFFFF
    For i = LBound(bytes) To UBound(bytes)
        crc = crc Xor CLng(bytes(i))
        For bit = 1 To 8
            low = (crc And 1) <> 0
            If crc < 0 Then
                crc = CLng(Fix((crc And &H7FFFFFFF) / 2)) Or &H40000000
            Else
                crc = CLng(Fix(crc / 2))
            End If
            If low Then crc = crc Xor &HEDB88320
        Next bit
    Next i
    ReferenceCrc = Right$("00000000" & Hex$(Not crc), 8)
End Function
